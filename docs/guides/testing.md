# Testing your code

Test code that posts ads without posting real ads. Give the client an `httpx.Client` with an `httpx.MockTransport`, and every request goes to a Python function of yours instead of tap.az.

## A complete test

Your code:

```python
from tapaz_sdk import TapAz


def post_flat(tap: TapAz, text: str, price: int) -> str:
    ad = tap.ads.create(635, text, properties={"740": "3722", "737": 85}, price=price, photo_ids=["p1"])
    return ad.url
```

The test:

```python
import json

import httpx
import pytest

from tapaz_sdk import InvalidRequestError, TapAz

CATEGORY = {
    "legacyResourceId": "635", "name": "Apartments", "hasChildren": False, "hasAutogeneratedTitle": True,
    "isPhotoRequiredOnCreate": True, "isPriceRequired": True,
    "properties": [
        {"legacyResourceId": "740", "name": "Listing type", "type": "CollectionProperty", "isRequired": True,
         "options": [{"legacyResourceId": "3722", "name": "For sale", "childrenPropertyIds": []}]},
        {"legacyResourceId": "737", "name": "Area", "type": "RangeProperty", "isRequired": True,
         "settings": {"from": 1, "to": 100000}, "options": []},
    ],
}
USER = {"legacyId": "77", "name": "Test", "email": "test@example.az", "phone": "+994501112233"}


def handler(request: httpx.Request) -> httpx.Response:
    body = json.loads(request.content)
    operation = body.get("operationName")
    if operation == "Category":
        return httpx.Response(200, json={"data": {"node": CATEGORY}})
    if operation == "CurrentUser":
        return httpx.Response(200, json={"data": {"currentUser": {"id": "VXNlcjo3Nw", **USER}}})
    if operation == "CreateAd":
        ad = {"legacyResourceId": 5002, "status": "PENDING", "path": "/elanlar/5002"}
        return httpx.Response(200, json={"data": {"createAd": {"entity": ad, "errors": []}}})
    return httpx.Response(400, json={"errors": [{"message": f"unexpected {operation}"}]})


def client() -> TapAz:
    return TapAz("_tapaz_session=test", max_retries=0, http_client=httpx.Client(transport=httpx.MockTransport(handler)))


def test_post_flat():
    assert post_flat(client(), "3 room flat", 150000) == "https://tap.az/elanlar/5002"


def test_area_out_of_range_is_caught_before_sending():
    tap = client()
    with pytest.raises(InvalidRequestError) as info:
        tap.ads.create(635, "flat", properties={"740": "3722", "737": 0}, price=1, photo_ids=["p1"])
    assert info.value.errors[0]["field"] == "properties.737"
```

`max_retries=0` keeps a failing test fast.

## What the answers look like

- GraphQL calls are POSTs to `https://tap.az/graphql` with `operationName` and `variables`. Route on `operationName`: `Categories`, `Category`, `PropertyOptions`, `Regions`, `ContactTypes`, `CurrentUser`, `Ad`, `MyAds`, `DeleteReasons` for reads; `CreateAd`, `UpdateAd`, `DestroyAd`, `ProlongAds`, `LoginUser` for writes.
- Answer reads with `{"data": {...}}` and writes with `{"data": {"createAd": {"entity": ..., "errors": []}}}`.
- Photo uploads are POSTs to `https://photos.tap.az/pond`; answer with `{"photos": [{"id": "abc"}]}`.
- The SMS login service is `https://api.digit-u.id/auth` and `/auth/verify`; answer the check with `{"access_token": "..."}`. The `LoginUser` answer must set a cookie with a `set-cookie` header for the sign in to succeed.

## Failures

```python
from tapaz_sdk import OutcomeUnknownError


def test_create_times_out():
    def slow(request):
        if json.loads(request.content).get("operationName") == "CreateAd":
            raise httpx.ReadTimeout("slow", request=request)
        return handler(request)

    tap = TapAz("_tapaz_session=test", max_retries=0, http_client=httpx.Client(transport=httpx.MockTransport(slow)))
    with pytest.raises(OutcomeUnknownError):
        tap.ads.create(635, "flat", properties={"740": "3722", "737": 85}, price=1, photo_ids=["p1"])
```

## Async

The same handler works with `httpx.AsyncClient(transport=httpx.MockTransport(handler))` and `AsyncTapAz`, under `pytest-asyncio`.

The SDK's own [tests/conftest.py](https://github.com/qiyascc/tapaz-python-sdk/blob/main/tests/conftest.py) has a fuller fake of tap.az, with sessions, categories and ads, that is worth reading before writing a large suite.
